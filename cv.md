### Code Examples:
Here is an example of my solution for the "Peak array index" KATA from CodeWars:

function peak(arr) {
  for (let i = 1; i < arr.length - 1; i++) {
    let leftsum = arr.slice(0, i).reduce((accumulator, currentvalue) => accumulator + currentvalue, 0);
    let rightsum = arr.slice(i + 1).reduce((accumulator, currentvalue) => accumulator + currentvalue, 0);

    if (leftsum === rightsum) {
      return i;
    }
  }
  return -1;
}
